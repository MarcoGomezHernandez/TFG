\section{Función de evaluación\label{SEC:CODIGOS_EVALUACION}}


\CPPCode[COD:BUTTERWORTH]{Filtro Butterworth paso bajo bidireccional}{Aplicación del filtro Butterworth paso bajo de cuarto orden en fase cero (\texttt{filtfilt}) sobre la señal con relleno por reflexión impar. La función \texttt{iir\_lowpass} diseña el filtro a la frecuencia de corte \texttt{fc} con frecuencia de muestreo \texttt{fs}, y \texttt{to\_sos} lo convierte a secciones de segundo orden para estabilidad numérica. La versión offline y online comparten esta lógica.}{offline/evaluation.tpp}{135}{137}{1}

\CPPCode[COD:PADDING]{Relleno por reflexión impar}{Implementación del relleno por reflexión impar en los extremos de la señal para mitigar los transitorios del filtro IIR. Se calcula la reflexión respecto al primer y último punto para garantizar continuidad en la derivada.}{online/evaluation.cpp}{91}{105}{1}

\CPPCode[COD:PEARSON]{Cálculo de la puntuación de forma mediante correlación de Pearson}{Correlación de Pearson entre la señal candidata y la referencia. La señal se centra restando su media, se calcula $r \in [-1, 1]$ y se normaliza a $[0, 1]$. La opción \texttt{search\_phase} invierte el resultado para buscar anti-correlación. Las operaciones vectorizadas (\texttt{mean}, \texttt{sum}, \texttt{sqr}) se realizan mediante KFR (Apéndice~\ref{SUBSEC:KFRREPO}).}{offline/evaluation.tpp}{61}{75}{1}

\CPPCode[COD:RANGESCORE]{Cálculo de la puntuación de rango}{Compara los extremos observados de la corriente candidata con los esperados. El error medio se normaliza por la distancia máxima admisible \texttt{pts\_dist\_max} y se devuelve como $1 - e_{\mathrm{norm}}$.}{offline/evaluation.tpp}{79}{89}{1}

\CPPCode[COD:RESCALE]{Funciones de reescalado de rangos al rango de corriente esperado}{La función \texttt{rescale\_to\_target} reescala con desplazamiento (usada para $I_{\mathrm{slow}}$, que hereda el nivel DC de $V_{\mathrm{pre}}$). La función \texttt{rescale\_to\_target\_no\_offset} reescala sin desplazamiento (usada para $I_{\mathrm{fast}}$, centrada en cero tras la sustracción de la componente filtrada).}{offline/evaluation.tpp}{38}{56}{1}
